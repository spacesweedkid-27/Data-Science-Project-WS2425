\subsection{Data Collection}
\begin{frame}{Data Collection}
    \begin{itemize}
        \item Rhythm
        \begin{enumerate}
            \item Retrieve duration \autocite{spot} and tempo \autocite{sbpm}
            \item Create tables
        \end{enumerate}
        \item Chords
        \begin{enumerate}
            \item Use chord-link finder to search for links on \textcite{ug}
            \item Run chord webscraper for found song links
            \item Create tables
            \item Scrape genres from \textcite{lastfm} and create table
        \end{enumerate}
        \item Lyrics
        \begin{enumerate}
            \item Retrieve links to lyrics from \textcite{lyr}
            \item Scrape lyrics and create tables
        \end{enumerate}
    \end{itemize}
\end{frame}

\section{What we found out}
\subsection{Results}
\begin{frame}{Results}
    \begin{itemize}
        \item Rhythm:
            \begin{itemize} 
                \item Duration: Overall downward trend
                \item Tempo: (Surprisingly) almost no changes
            \end{itemize}
        \item Chords:
            \begin{itemize} 
                \item Most common chords are C Major and A Minor
                \item Most common harmonic progression is I, VI, III, VII
                \item Most frequent genre is pop-music
                \item Higher variance indicates more musical variety
            \end{itemize}
        \item Lyrics:
            \begin{itemize} 
                \item Most common words are "oh", "yeah", "la"
            \end{itemize}
    \end{itemize}
\end{frame}
