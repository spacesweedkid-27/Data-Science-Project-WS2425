\section{What we wanted to know}
\begin{frame}[allowframebreaks]{What we wanted to know}
    \begin{block}{Generally}
    How did various musical/lyrical elements vary in the most popular songs of the last 20 years?
    \end{block}

    \framebreak

    \begin{block}{Subquestions}
    \onehalfspacing
    \begin{itemize}
        \item Rhythm \begin{itemize}
                \item Duration
                \item Tempo
            \end{itemize}
        \item Chords \begin{itemize}
                \item Chord-progressions
                \item Chords and chord types
                \item Intervals between chords
            \end{itemize}
        \item Lyrics \begin{itemize}
                \item Words and combinations
                \item Polarity
            \end{itemize}
    \end{itemize}
    \end{block}
\end{frame}

\section{How we found out}
\subsection{Data Sources}
\begin{frame}{Data Sources}
    \begin{itemize}
        \item \textbf{Main source:} \textcite{bill} of 2005 up to 2024
        \item \textbf{Refined sources using songs from the main source:}
            \begin{itemize}
                \item Rhythm: \textcite{spot}, \textcite{sbpm}
                \item Chords: \textcite{ug}
                \item Lyrics: \textcite{lastfm}, \textcite{lyr}
                
            \end{itemize}
        \end{itemize}
\end{frame}
\subsection{Data Collection}
\begin{frame}{Data Collection}
    \begin{itemize}
        \item Rhythm
        \begin{enumerate}
            \item Retrieve duration \autocite{spot} and tempo \autocite{sbpm}
            \item Create tables
        \end{enumerate}
        \item Chords
        \begin{enumerate}
            \item Use chord-link finder to search for links on \textcite{ug}
            \item Run chord webscraper for found song links
            \item Create tables
            \item Scrape genres from \textcite{lastfm} and create table
        \end{enumerate}
        \item Lyrics
        \begin{enumerate}
            \item Retrieve links to lyrics from \textcite{lyr}
            \item Scrape lyrics and create tables
        \end{enumerate}
    \end{itemize}
\end{frame}
\subsection{Challenges}
\begin{frame}{Challenges}
    \begin{itemize}
        \item No public APIs
        \item Licensing restrictions
        \item Incorrect search results
        \item URL changes
        \item Inconsistent artist names
            \begin{itemize} 
                \item Featured artists
                \item Special characters
            \end{itemize}
    \end{itemize}
\end{frame}
